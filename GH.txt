-- ── UTILITY ─────────────────────────────────────────────────────────────────
local function GetChar(p)   return p.Character end
local function GetHum(p)    local c=GetChar(p); return c and c:FindFirstChildOfClass("Humanoid") end
local function IsAlive(p)   local h=GetHum(p);  return h and h.Health > 0 end
local function GetRoot(p)   local c=GetChar(p); return c and c:FindFirstChild("HumanoidRootPart") end
local function GetHead(p)   local c=GetChar(p); return c and c:FindFirstChild("Head") end

local function IsAlly(p)
    if not Config.Aimbot.TeamCheck then return false end
    return p.Team == LocalPlayer.Team
end

local function W2S(pos)
    local sp, on = Camera:WorldToViewportPoint(pos)
    return Vector2.new(sp.X, sp.Y), on, sp.Z
end

local function Dist(p)
    local r, mr = GetRoot(p), GetRoot(LocalPlayer)
    if not r or not mr then return math.huge end
    return math.floor((r.Position - mr.Position).Magnitude)
end

local function GetFruit(p)
    local char = GetChar(p)
    if not char then return nil end
    local bp = p:FindFirstChild("Backpack")
    local sources = {char, bp}
    for _, src in pairs(sources) do
        if src then
            for _, tool in pairs(src:GetChildren()) do
                if tool:IsA("Tool") then
                    for _, f in pairs(FRUITS) do
                        if tool.Name:find(f) then return f end
                    end
                end
            end
        end
    end
    return nil
end

local function GetStyle(p)
    local char = GetChar(p)
    if not char then return nil end
    for _, tool in pairs(char:GetChildren()) do
        if tool:IsA("Tool") then
            for _, s in pairs(STYLES) do
                if tool.Name:find(s) then return s end
            end
        end
    end
    return nil
end

local function IsVisible(p)
    local head = GetHead(p)
    local char = GetChar(p)
    if not head or not char then return false end
    local origin = Camera.CFrame.Position
    local dir    = (head.Position - origin)
    local ray    = Ray.new(origin, dir)
    local hit    = workspace:FindPartOnRayWithIgnoreList(ray, {
        LocalPlayer.Character, workspace.Terrain
    })
    return hit ~= nil and hit:IsDescendantOf(char)
end

local function GetTarget()
    local vp     = Camera.ViewportSize
    local cx, cy = vp.X/2, vp.Y/2
    local best, bestD = nil, Config.Aimbot.FOV

    for _, p in pairs(Players:GetPlayers()) do
        if p == LocalPlayer or IsAlly(p) or not IsAlive(p) then continue end
        local char = GetChar(p)
        local part = char and char:FindFirstChild(Config.Aimbot.HitPart)
        if not part then continue end
        local sp, on = W2S(part.Position)
        if not on then continue end
        local d = math.sqrt((sp.X-cx)^2 + (sp.Y-cy)^2)
        if d < bestD then bestD=d; best=p end
    end
    return best
end

local function NewDraw(t) return Drawing.new(t) end

local function GetESP(p)
    if ESPPool[p] then return ESPPool[p] end
    local e = {
        Box        = NewDraw("Square"),
        HpBG       = NewDraw("Square"),
        HpFill     = NewDraw("Square"),
        Name       = NewDraw("Text"),
        Dist       = NewDraw("Text"),
        Fruit      = NewDraw("Text"),
        Style      = NewDraw("Text"),
        Snap       = NewDraw("Line"),
        HDot       = NewDraw("Circle"),
        Skel       = {},
    }
    -- defaults
    e.Box.Thickness=1; e.Box.Filled=false
    e.HpBG.Filled=true; e.HpFill.Filled=true
    for _, lbl in pairs({e.Name,e.Dist,e.Fruit,e.Style}) do
        lbl.Size=13; lbl.Center=true; lbl.Outline=true
    end
    e.Dist.Size=12; e.Fruit.Size=12; e.Style.Size=11
    e.HDot.Filled=false; e.HDot.Thickness=1; e.HDot.NumSides=24
    e.Snap.Thickness=1
    for i=1,#BONE_PAIRS do
        local l=NewDraw("Line"); l.Thickness=1; e.Skel[i]=l
    end
    ESPPool[p]=e
    return e
end

local function HideAll(e)
    for _, v in pairs(e) do
        if type(v)=="userdata" then v.Visible=false
        elseif type(v)=="table" then for _,l in pairs(v) do pcall(function() l.Visible=false end) end
        end
    end
end

local function GetBounds(p)
    local char = GetChar(p)
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    local head = char:FindFirstChild("Head")
    if not root or not head then return nil end
    local rSP,rOn = W2S(root.Position)
    local hSP,hOn = W2S(head.Position)
    if not rOn and not hOn then return nil end
    local topY    = hSP.Y - 10
    local botY    = rSP.Y + 15
    local h       = botY - topY
    local w       = h * 0.55
    local cx      = (hSP.X+rSP.X)/2
    return { x=cx-w/2, y=topY, w=w, h=h, hSP=hSP, rSP=rSP }
end

local function UpdateESP()
    if not Config.ESP.Enabled then
        for _,e in pairs(ESPPool) do HideAll(e) end
        return
    end

    local vp = Camera.ViewportSize

    for _, p in pairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local e = GetESP(p)

        if not IsAlive(p) then HideAll(e); continue end

        local b = GetBounds(p)
        if not b then HideAll(e); continue end

        local hum     = GetHum(p)
        local hp      = hum and hum.Health    or 0
        local maxHp   = hum and hum.MaxHealth or 100
        local ratio   = math.clamp(hp/maxHp, 0, 1)
        local vis     = IsVisible(p)
        local boxCol  = vis and Config.Colors.Visible or Config.Colors.Invisible

        -- 1 Box
        if Config.ESP.Boxes then
            e.Box.Visible=true; e.Box.Color=boxCol
            e.Box.Position=Vector2.new(b.x,b.y)
            e.Box.Size=Vector2.new(b.w,b.h)
        else e.Box.Visible=false end

        -- 2 Health bar
        if Config.ESP.HealthBar then
            local bx=b.x-7
            e.HpBG.Visible=true
            e.HpBG.Position=Vector2.new(bx,b.y)
            e.HpBG.Size=Vector2.new(4,b.h)
            e.HpBG.Color=Color3.fromRGB(20,20,20)
            local fh=b.h*ratio
            e.HpFill.Visible=true
            e.HpFill.Position=Vector2.new(bx, b.y+b.h-fh)
            e.HpFill.Size=Vector2.new(4,fh)
            e.HpFill.Color=Color3.fromRGB(
                math.floor(255*(1-ratio)),
                math.floor(255*ratio), 40)
        else e.HpBG.Visible=false; e.HpFill.Visible=false end

        -- 3 Name
        if Config.ESP.NameTag then
            e.Name.Visible=true
            e.Name.Text=p.Name
            e.Name.Position=Vector2.new(b.x+b.w/2, b.y-16)
            e.Name.Color=Config.Colors.Text
        else e.Name.Visible=false end

        -- 4 Distance
        if Config.ESP.Distance then
            e.Dist.Visible=true
            e.Dist.Text=Dist(p).."m"
            e.Dist.Position=Vector2.new(b.x+b.w/2, b.y+b.h+2)
            e.Dist.Color=Color3.fromRGB(180,180,180)
        else e.Dist.Visible=false end

        -- 5 Fruit ESP
        if Config.ESP.FruitESP then
            local fr = GetFruit(p)
            if fr then
                e.Fruit.Visible=true
                e.Fruit.Text="["..fr.."]"
                e.Fruit.Position=Vector2.new(b.x+b.w/2, b.y-28)
                e.Fruit.Color=Config.Colors.Fruit
            else e.Fruit.Visible=false end
        else e.Fruit.Visible=false end

        -- 6 Snap lines
        if Config.ESP.SnapLines then
            e.Snap.Visible=true
            e.Snap.From=Vector2.new(vp.X/2, vp.Y)
            e.Snap.To=Vector2.new(b.x+b.w/2, b.y+b.h)
            e.Snap.Color=boxCol
        else e.Snap.Visible=false end

        -- 7 Skeleton
        if Config.ESP.Skeleton then
            local char = GetChar(p)
            for i, pair in pairs(BONE_PAIRS) do
                local l=e.Skel[i]
                local b1=char and char:FindFirstChild(pair[1])
                local b2=char and char:FindFirstChild(pair[2])
                if b1 and b2 then
                    local p1,o1=W2S(b1.Position)
                    local p2,o2=W2S(b2.Position)
                    if o1 or o2 then
                        l.Visible=true; l.From=p1; l.To=p2; l.Color=boxCol
                    else l.Visible=false end
                else l.Visible=false end
            end
        else for _,l in pairs(e.Skel) do l.Visible=false end end

        -- 8 Fighting style
        if Config.ESP.StyleLabel then
            local st=GetStyle(p)
            if st then
                e.Style.Visible=true
                e.Style.Text=st
                e.Style.Position=Vector2.new(b.x+b.w/2, b.y+b.h+14)
                e.Style.Color=Config.Colors.Style
            else e.Style.Visible=false end
        else e.Style.Visible=false end

        -- 9 Vis check — already baked into boxCol above, extra dot indicator
        -- (color switches green/red live on every drawn element)

        -- 10 Head dot
        if Config.ESP.HeadDot then
            local hd=GetHead(p)
            if hd then
                local sp,on=W2S(hd.Position)
                if on then
                    e.HDot.Visible=true; e.HDot.Position=sp
                    e.HDot.Radius=5; e.HDot.Color=boxCol
                else e.HDot.Visible=false end
            else e.HDot.Visible=false end
        else e.HDot.Visible=false end
    end

    -- Cleanup left players
    for p, e in pairs(ESPPool) do
        if not Players:FindFirstChild(p.Name) then
            HideAll(e)
            for _, v in pairs(e) do
                if type(v)=="userdata" then pcall(function() v:Remove() end)
                elseif type(v)=="table" then for _,l in pairs(v) do pcall(function() l:Remove() end) end end
            end
            ESPPool[p]=nil
        end
    end
end

-- ── RENDER LOOP ─────────────────────────────────────────────────────────────
RunService.RenderStepped:Connect(function()
    local vp=Camera.ViewportSize
    local cx,cy=vp.X/2,vp.Y/2

    -- FOV circle
    FOVDraw.Visible    = Config.Aimbot.FOVCircle
    FOVDraw.Position   = Vector2.new(cx,cy)
    FOVDraw.Radius     = Config.Aimbot.FOV

    -- Aimbot — hold RMB
    if Config.Aimbot.Enabled and RMBHeld then
        local target = GetTarget()
        if target then
            local char = GetChar(target)
            local part = char and char:FindFirstChild(Config.Aimbot.HitPart)
            if part then
                local chance = Config.Aimbot.HitChance or 100
                if math.random(1,100) <= chance then
                    local cur  = Camera.CFrame
                    local goal = CFrame.new(cur.Position, part.Position)
                    Camera.CFrame = cur:Lerp(goal, Config.Aimbot.Smoothing)
                end
            end
        end
    end

    -- ESP
    UpdateESP()

    -- Crosshair
    if Config.Visuals.Crosshair then
        local s=Config.Visuals.CrosshairSize
        CHLines[1].Visible=true; CHLines[1].From=Vector2.new(cx-s,cy);   CHLines[1].To=Vector2.new(cx+s,cy)
        CHLines[2].Visible=true; CHLines[2].From=Vector2.new(cx,cy-s);   CHLines[2].To=Vector2.new(cx,cy+s)
        CHLines[3].Visible=false; CHLines[4].Visible=false
    else
        for _,l in pairs(CHLines) do l.Visible=false end
    end
end)
